One-Click Supabase SQL Setup & RLS Unlock Script

If your tables say "RLS Denied" or do not save data, copy this script, go to your Supabase SQL Editor ↗, paste and click Run. This enables unrestricted read & write for this batch console. 

-- Enable unrestricted read/write for BNSCK Batch Portal tables
ALTER TABLE IF EXISTS gallery DISABLE ROW LEVEL SECURITY;
ALTER TABLE IF EXISTS timeline DISABLE ROW LEVEL SECURITY;
ALTER TABLE IF EXISTS notices DISABLE ROW LEVEL SECURITY;
ALTER TABLE IF EXISTS teachers DISABLE ROW LEVEL SECURITY;
ALTER TABLE IF EXISTS students DISABLE ROW LEVEL SECURITY;
ALTER TABLE IF EXISTS capsule DISABLE ROW LEVEL SECURITY;

-- Reset primary key auto-increment sequences to prevent duplicate key errors
SELECT setval(pg_get_serial_sequence('gallery', 'id'), COALESCE((SELECT MAX(id) FROM gallery), 0) + 1, false);
SELECT setval(pg_get_serial_sequence('timeline', 'id'), COALESCE((SELECT MAX(id) FROM timeline), 0) + 1, false);
SELECT setval(pg_get_serial_sequence('teachers', 'id'), COALESCE((SELECT MAX(id) FROM teachers), 0) + 1, false);
SELECT setval(pg_get_serial_sequence('notices', 'id'), COALESCE((SELECT MAX(id) FROM notices), 0) + 1, false);
SELECT setval(pg_get_serial_sequence('capsule', 'id'), COALESCE((SELECT MAX(id) FROM capsule), 0) + 1, false);

-- Ensure gallery table has all needed columns
CREATE TABLE IF NOT EXISTS gallery (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  title TEXT NOT NULL,
  url TEXT NOT NULL,
  tag TEXT DEFAULT 'CAMPUS',
  description TEXT DEFAULT '',
  display_order INT DEFAULT 1,
  created_at TIMESTAMPTZ DEFAULT NOW()
);
